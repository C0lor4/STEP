\textbf{(i)} $\ds\frac{dy}{dx}=v+x\frac{dv}{dx}$

$$xy\frac{dy}{dx}+y^2-2x^2=0$$
$$x(xv)\left(v+x\frac{dv}{dx}\right)+x^2v^2-2x^2=0$$
$$x^2v^2+x^3v\frac{dv}{dx}+x^2v^2-2x^2=0$$
$$\implies xv\frac{dv}{dx}+2v^2-2=0$$

\begin{align*}
xv\frac{dv}{dx}+2v^2-2&=0\\
xv\frac{dv}{dx}&=2(1-v^2)\\
\frac{v}{1-v^2}\frac{dv}{dx}&=\frac2x
\end{align*}

$$\int \frac{v}{1-v^2}\,dv=\int \frac2x\,dx$$

$$-\frac12\ln|1-v^2|=2\ln|x|+C$$

$$\ln|x^4(1-v^2)|=C$$

$$x^4(v^2-1)=C$$

$$x^4\left(\frac{y^2}{x^2}-1\right)=C$$

$$\implies x^2(y^2-x^2)=C$$

\bigskip\hrule\bigskip

\textbf{(ii)} Let $\ds y=xv,\ \frac{dy}{dx}=v+x\frac{dv}{dx}$

$$y\frac{dy}{dx}+6x+5y=0$$
$$xv\left(v+x\frac{dv}{dx}\right)+6x+5xv=0$$
$$xv\frac{dv}{dx}+(v+2)(v+3)=0$$

\begin{align*}
\int \frac{v}{(v+2)(v+3)}\,dv
&=-\int\frac1x\,dx\\
\int\left(-\frac2{v+2}+\frac3{v+3}\right)\,dv
&=-\ln|x|+C\\
-2\ln|v+2|+3\ln|v+3|
&=-\ln|x|+C\\
\ln\left|x\frac{(v+3)^3}{(v+2)^2}\right|&=C
\end{align*}
$$\frac{(y+3x)^3}{(y+2x)^2}=C$$
