\begin{align*}
\sum_{n=1}^{\infty}P(X\geq n)
&=\sum_{n=1}^{\infty}\sum_{r=n}^{\infty}P(X=r)\\
&=\sum_{r=1}^{\infty}\sum_{n=1}^{r}P(X=r)\\
&=\sum_{r=1}^{\infty}rP(X=r)\\
&=\mathrm{E}(X)
\end{align*}

\bigskip\hrule\bigskip

$$X\geq4\iff \text{first three boxes are all daddy or all mummy}$$

$$P(X\geq4)=p^3+q^3$$

$$X\geq n\iff \text{first }n-1\text{ boxes are all daddy or all mummy}$$

$$P(X\geq n)=p^{n-1}+q^{n-1}$$

$$P(X\geq1)=1,\  P(X\geq2)=1$$
\begin{align*}
\mathrm{E}(X)
&=1+1+\sum_{n=3}^{\infty}\bigl(p^{n-1}+q^{n-1}\bigr)\\
&=2+\frac{p^2}{1-p}+\frac{q^2}{1-q}\\
&=2+\frac{p^2}{q}+\frac{q^2}{p}\\
&=2+\frac{p^3+q^3}{pq}
\end{align*}

$$p+q=1\implies p^3+q^3=(p+q)^3-3pq(p+q)=1-3pq$$

$$\therefore \mathrm{E}(X)=2+\frac{1-3pq}{pq}=\frac1{pq}-1$$

$$pq\leq\left(\frac{p+q}{2}\right)^2=\frac14$$

$$\therefore \mathrm{E}(X)=\frac1{pq}-1\geq4-1=3$$
