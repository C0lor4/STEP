	\section*{Problem 4.}
	\textbf{i)}
	$$
	\begin{aligned}
	\text{LHS}&=\int_{0}^{\infty}e^{-st}\;e^{-bt}f(t)\;dt=\int_{0}^{\infty}e^{-(s+b)t}f(t)\;dt\\
	\text{RHS}&=F(s+b)=\int_{0}^{\infty}e^{-(s+b)t}f(t)\;dt\\
	\therefore\text{LHS}&=\text{RHS}
	\end{aligned}
	$$

	\textbf{ii)}
	$$
	\begin{aligned}
		\text{LHS}&=\int_{0}^{\infty}e^{-st}f(at)\;dt\\
		\text{Assume } u=at\text{, }\frac{du}{dt}=a\\
		\text{LHS}&=\int_{0}^{\infty}e^{-(s/a)\;u}f(u)\times a^{-1}\;du\\
		\text{RHS}&=a^{-1}F(\frac{s}{a})=a^{-1}\;\int_{0}^{\infty}e^{-(s/a)\;t}f(t)\;dt\\
		\therefore \text{LHS}&=\text{RHS}
	\end{aligned}
	$$
