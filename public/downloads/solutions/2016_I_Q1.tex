\textbf{(i)}

\text{When } n=1:
$$\quad p_1(x)=(x+1)^2-3x=x^2-x+1$$
$$\implies (x+1)p_1(x)=(x+1)(x^2-x+1)=x^3+1=(x+1)q_1(x)$$
$$\implies p_1(x)=q_1(x)$$

\bigskip
\text{When } n=2:
$$\quad p_2(x)=(x+1)^4-5x(x^2+x+1)$$
$$\implies p_2(x)=(x^4+4x^3+6x^2+4x+1)-(5x^3+5x^2+5x)=x^4-x^3+x^2-x+1$$
$$\implies (x+1)p_2(x)=(x+1)(x^4-x^3+x^2-x+1)=x^5+1=(x+1)q_2(x)$$
$$\implies p_2(x)=q_2(x)$$

\bigskip
\text{When } n=3:
$$\quad p_3(x)=(x+1)^6-7x(x^2+x+1)^2$$
$$\implies p_3(x)=(x^6+6x^5+15x^4+20x^3+15x^2+6x+1)-(7x^5+14x^4+21x^3+14x^2+7x)$$
$$\implies p_3(x)=x^6-x^5+x^4-x^3+x^2-x+1$$
$$\implies (x+1)p_3(x)=(x+1)(x^6-x^5+x^4-x^3+x^2-x+1)=x^7+1=(x+1)q_3(x)$$
$$\implies p_3(x)=q_3(x)$$

\bigskip
\text{When } n=4:
$$\quad p_4(x)=(x+1)^8-9x(x^2+x+1)^3$$

\text{Let } x=1:
$$\quad p_4(1)=2^8-9\cdot1\cdot3^3=256-243=13$$
$$q_4(1)=1^8-1^7+1^6-1^5+1^4-1^3+1^2-1+1=1$$
$$\implies p_4(1)\ne q_4(1)$$
$$\implies p_4(x)\ne q_4(x)$$
