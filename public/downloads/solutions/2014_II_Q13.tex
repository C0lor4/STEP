$\bigl( \text{First num.} \neq \text{Second num.} \bigr)\times \bigl( \text{Third num.} \neq \text{First two num.} \bigr)\times \bigl( \text{Fourth num.} = \text{One of the num.} \bigr)$

$$P(X=4)=\left(1-\frac1n\right)\left(1-\frac2n\right)\frac3n$$

\bigskip\hrule\bigskip

\textbf{(i)}
$$P(X=r)=\left(1-\frac1n\right)\left(1-\frac2n\right)\cdots\left(1-\frac{r-2}{n}\right)\frac{r-1}{n}$$

where $2\leq r\leq n+1$

$\because X\in\{2,3,\dots,n+1\}$, the sum of the probability = 1

$$\therefore \sum_{r=2}^{n+1}P(X=r)=1$$

$$\therefore \frac1n+\left(1-\frac1n\right)\frac2n+\left(1-\frac1n\right)\left(1-\frac2n\right)\frac3n+\cdots=1$$

\bigskip\hrule\bigskip

\textbf{(ii)}

$$\boxed{\mathrm{E}(X)=\sum_{r=2}^{n+1}r\left(1-\frac1n\right)\left(1-\frac2n\right)\cdots\left(1-\frac{r-2}{n}\right)\frac{r-1}{n}}$$

\bigskip\hrule\bigskip

\textbf{(iii)} $\ds X\geq k\iff I_1,I_2,\dots,I_{k-1}$ are distinct

$$\boxed{P(X\geq k)=\left(1-\frac1n\right)\left(1-\frac2n\right)\cdots\left(1-\frac{k-2}{n}\right)}$$

\bigskip\hrule\bigskip

\textbf{(iv)}

\begin{align*}
\mathrm{E}(Y)
&=\sum_{r=1}^{N}rP(Y=r)\\
&=\sum_{r=1}^{N}\sum_{k=1}^{r}P(Y=r)\\
&=\sum_{k=1}^{N}\sum_{r=k}^{N}P(Y=r)\\
&=\sum_{k=1}^{N}P(Y\geq k)
\end{align*}

Let $\ds S_k=\left(1-\frac1n\right)\left(1-\frac2n\right)\cdots\left(1-\frac{k-1}{n}\right)$, where $S_1=1$

From part (iii) $\implies \ds P(X\geq k+1)=S_k,\quad 1\leq k\leq n$

\begin{align*}
\mathrm{E}(X)
&=\sum_{k=1}^{n+1}P(X\geq k)\\
&=P(X\geq1)+\sum_{k=1}^{n}P(X\geq k+1)\\
&=1+\sum_{k=1}^{n}S_k
\end{align*}

From part (ii) $\implies \ds P(X=k+1)=S_k\,\frac{k}{n},\quad 1\leq k\leq n$

$\because \ds \sum_{k=1}^{n}P(X=k+1)=1\quad \therefore \sum_{k=1}^{n}\frac{k}{n}S_k=1$

\begin{align*}
\mathrm{E}(X)
&=\sum_{k=1}^{n}(k+1)P(X=k+1)\\
&=\sum_{k=1}^{n}(k+1)\frac{k}{n}S_k\\
&=\sum_{k=1}^{n}\frac{k^2}{n}S_k+\sum_{k=1}^{n}\frac{k}{n}S_k\\
&=1+\sum_{k=1}^{n}\frac{k^2}{n}S_k
\end{align*}


$$1+\sum_{k=1}^{n}S_k=1+\sum_{k=1}^{n}\frac{k^2}{n}S_k$$

$$\therefore \sum_{k=1}^{n}S_k\left(1-\frac{k^2}{n}\right)=0$$

$$\boxed{\left(1-\frac{1^2}{n}\right)+\left(1-\frac1n\right)\left(1-\frac{2^2}{n}\right)+\left(1-\frac1n\right)\left(1-\frac2n\right)\left(1-\frac{3^2}{n}\right)+\cdots=0}$$
