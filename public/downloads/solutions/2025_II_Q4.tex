\textbf{(i)} Let $\ds a=\fl{x},\  b=x-\fl{x}$
$$x=a+b,\qquad 0\leq b<1$$
\begin{align*}
\because n\in \mathbb{Z^+}\\
\therefore\fl{x+n}
&=\fl{a+b+n}\\
&=a+n\\
&=\fl{x}+n
\end{align*}
\bigskip\hrule\bigskip

\textbf{(ii)(a)}
\begin{align*}
f_n\left(x+\frac1n\right)
&=\fl{x+\frac1n}+\fl{x+\frac2n}+\cdots+\fl{x+1}-\fl{nx+1}\\
&=\fl{x+\frac1n}+\fl{x+\frac2n}+\cdots+\fl{x+\frac{n-1}{n}}
+\fl{x}+1-\fl{nx}-1\\
&=\fl{x}+\fl{x+\frac1n}+\cdots+\fl{x+\frac{n-1}{n}}-\fl{nx}\\
&=f_n(x)
\end{align*}

$$\boxed{f_n\left(x+\frac1n\right)=f_n(x)}$$

\bigskip\hrule\bigskip

\textbf{(ii)(b)} If  $0\leq t<\dfrac{1}{n}\implies 0\leq t<1$

For $0\leq t<\dfrac1n\implies 0\leq nt<1\implies\lfloor nt\rfloor=0$

For $\ds r=0,1,\dots,n-1 \implies  0\leq t+\frac rn<\frac1n+\frac rn=\frac{r+1}{n}\leq1$

$\ds\because t<\dfrac1n\quad \therefore  t+\frac rn<1 \implies \left\lfloor t+\frac rn\right\rfloor=0$

$$\implies \boxed{f_n(t)=0}$$

\bigskip\hrule\bigskip



\textbf{(ii)(c)} Let $\ds t=x-\frac{k}{n} \text{ and } 0\leq x-\frac{k}{n}<\frac1n\implies x=t+\dfrac{k}{n}$

From part(ii)(a)$\ds \implies f_n(x)=f_n\left(t+\frac{k}{n}\right)=f_n(t)$

$$\because f_n(t)=0 \therefore \boxed{f_n(x)=0}$$

\bigskip\hrule\bigskip

\textbf{(iii)(a)} When $n=2$ for part (ii)(c)

$$f_2(u)=\fl{u}+\fl{u+\frac12}-\fl{2u}=0$$

Let $\ds u=\dfrac{x}{2} \quad \implies \fl{\frac{x}{2}}+\fl{\frac{x+1}{2}}-\fl{x}=0 \implies \boxed{\fl{\frac{x}{2}}+\fl{\frac{x+1}{2}}=\fl{x}}$

\bigskip\hrule\bigskip

\textbf{(iii)(b)} Let
$$S=\fl{\frac{x+1}{2}}+\fl{\frac{x+2}{2^2}}+\cdots+\fl{\frac{x+2^k}{2^{k+1}}}+\cdots$$

When $x=\dfrac{u}{2^k}$,  from part (iii)(a),
$$\fl{\frac{u}{2^{k+1}}}+\fl{\frac{u+2^k}{2^{k+1}}}=\fl{\frac{u}{2^k}} \implies \fl{\frac{u+2^k}{2^{k+1}}}=\fl{\frac{u}{2^k}}-\fl{\frac{u}{2^{k+1}}}$$
\begin{align*}
\therefore S_N
&=\sum_{k=0}^{N}\fl{\frac{x+2^k}{2^{k+1}}}\\
&=\sum_{k=0}^{N}\left(\fl{\frac{x}{2^k}}-\fl{\frac{x}{2^{k+1}}}\right)\\
&=\fl{x}-\fl{\frac{x}{2^{N+1}}}
\end{align*}

As $N\to\infty$, $\implies\left\lfloor\frac{x}{2^{N+1}}\right\rfloor=0$
$$\implies\boxed{S=\lfloor x\rfloor}$$
