\[
\begin{aligned}
\cos3x&=\cos(2x+x)\\
&=\cos2x\cos x-\sin2x\sin x\\
&=(2\cos^2x-1)\cos x-2\sin^2x\cos x\\
&=(2\cos^2x-1)\cos x-2(1-\cos^2x)\cos x
\end{aligned}
\]

\[
\boxed{\cos3x=4\cos^3x-3\cos x}
\]

\[
\begin{aligned}
\sin3x&=\sin(2x+x)\\
&=\sin2x\cos x+\cos2x\sin x\\
&=2\sin x\cos^2x+(1-2\sin^2x)\sin x\\
&=2\sin x(1-\sin^2x)+(1-2\sin^2x)\sin x
\end{aligned}
\]

\[
\boxed{\sin3x=3\sin x-4\sin^3x}
\]

\bigskip\hrule\bigskip

\textbf{(i)}

$$
7\sin x-8\sin^3x
=\sin x+2(3\sin x-4\sin^3x)
$$

$$
=\sin x+2\sin3x
$$

$$
I(\alpha)=\int_0^\alpha(\sin x+2\sin3x)\,dx
$$

$$
=\left[-\cos x-\frac23\cos3x\right]_0^\alpha
$$

$$
=-\cos\alpha-\frac23\cos3\alpha+\frac53
$$

Let $c=\cos\alpha$

$$
\cos3\alpha=4c^3-3c
$$

$$
I(\alpha)=-c-\frac23(4c^3-3c)+\frac53
$$

$$
\boxed{I(\alpha)=-\frac83c^3+c+\frac53}
$$

For $c=1$

$$
I(\alpha)=-\frac83+1+\frac53=0
$$

$$
\boxed{c=1}
$$

\bigskip\hrule\bigskip

\textbf{(ii)}

Eustace obtains

$$
I_E(\alpha)
=\left[\frac72\sin^2x-2\sin^4x\right]_0^\alpha
$$

$$
=\frac72\sin^2\alpha-2\sin^4\alpha
$$

$$
=\frac72(1-c^2)-2(1-c^2)^2
$$

$$
=\frac32+\frac12c^2-2c^4
$$

For $\cos\beta=-\dfrac16$

$$
I(\beta)
=-\frac83\left(-\frac16\right)^3-\frac16+\frac53
$$

$$
=\frac{245}{162}
$$

$$
I_E(\beta)
=\frac32+\frac12\left(-\frac16\right)^2
-2\left(-\frac16\right)^4
$$

$$
=\frac{245}{162}
$$

$$
\boxed{I_E(\beta)=I(\beta)}
$$

For the correct value of $I(\alpha)$

$$
\frac32+\frac12c^2-2c^4
=-\frac83c^3+c+\frac53
$$

$$
12c^4-16c^3-3c^2+6c+1=0
$$

$$
(c-1)^2(2c+1)(6c+1)=0
$$

$$
c=1,\qquad c=-\frac12,\qquad c=-\frac16
$$

Since $c=\cos\alpha$

$$
\boxed{
\alpha=2k\pi,\qquad
\alpha=2k\pi\pm\frac{2\pi}{3},\qquad
\alpha=2k\pi\pm\arccos\left(-\frac16\right),
\quad k\in\mathbb Z
}
$$
