\textbf{(i)}

$$\frac{x^2}{x+1}=x-1+\frac1{x+1}$$

\begin{align*}
\int_{1/m}^{m}\frac{x^2}{x+1}\,dx
&=\left[\frac{x^2}{2}-x+\ln(x+1)\right]_{1/m}^{m}\\
&=\frac{m^2}{2}-m-\frac1{2m^2}+\frac1m+\ln\frac{m+1}{1+\frac1m}\\
&=\frac{(m-1)^3(m+1)}{2m^2}+\ln m
\end{align*}

\textbf{(ii)} Let $\ds u=\frac1x,\ dx=-\frac1{u^2}\,du$

$$\int_{1/m}^{m}\frac{1}{x^n(x+1)}\,dx=\int_{m}^{1/m}\frac{u^{n+1}}{u+1}\left(-\frac1{u^2}\right)\,du\\=\int_{1/m}^{m}\frac{u^{n-1}}{u+1}\,du$$


\bigskip\hrule\bigskip

\textbf{(iii)(a)}

\begin{align*}
\int_{1/2}^{2}\frac{x^5+3}{x^3(x+1)}\,dx
&=\int_{1/2}^{2}\frac{x^2}{x+1}\,dx
+3\int_{1/2}^{2}\frac1{x^3(x+1)}\,dx\\
&=\int_{1/2}^{2}\frac{x^2}{x+1}\,dx
+3\int_{1/2}^{2}\frac{x^2}{x+1}\,dx\\
&=4\left(\frac38+\ln2\right)\\
&=\frac32+4\ln2
\end{align*}

\bigskip\hrule\bigskip

\textbf{(iii)(b)}

\begin{align*}
\int_{1}^{2}\frac{x^5+x^3+1}{x^3(x+1)}\,dx
&=\int_{1}^{2}\frac{x^2}{x+1}\,dx
+\int_{1}^{2}\frac1{x+1}\,dx
+\int_{1}^{2}\frac1{x^3(x+1)}\,dx
\end{align*}

Let $u=1/x,\  du=-x^{-2}dx=u^2dx$

$$\int_{1}^{2}\frac1{x^3(x+1)}\,dx=\int_{1/2}^{1}\frac{u^2}{u+1}\,du$$

\begin{align*}
\int_{1}^{2}\frac{x^5+x^3+1}{x^3(x+1)}\,dx
&=\int_{1/2}^{2}\frac{x^2}{x+1}\,dx+\int_{1}^{2}\frac1{x+1}\,dx\\
&=\frac38+\ln2+\ln\frac32\\
&=\frac38+\ln3
\end{align*}
