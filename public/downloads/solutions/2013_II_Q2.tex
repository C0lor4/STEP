\textbf{(i)} Let $\ds t=1-x, \ dx=-dt,\ 
\begin{array}{c|cc}
x&0&1\\
\hline
t&1&0
\end{array}
$


\begin{align*}
\int_0^1 x^{n-1}(1-x)^n\,dx
&=\int_1^0 (1-t)^{n-1}t^n(-dt)\\
&=\int_0^1 t^n(1-t)^{n-1}\,dt\\
&=\int_0^1 x^n(1-x)^{n-1}\,dx
\end{align*}

\begin{align*}
I_{n-1}
&=\int_0^1 x^{n-1}(1-x)^{n-1}\,dx\\
&=\int_0^1 x^{n-1}(1-x)^{n-1}\bigl(x+1-x\bigr)\,dx\\
&=\int_0^1 x^n(1-x)^{n-1}\,dx+\int_0^1 x^{n-1}(1-x)^n\,dx\\
&=2\int_0^1 x^{n-1}(1-x)^n\,dx
\end{align*}


$$u=x^n,\  dv=(1-x)^n\,dx,\ du=nx^{n-1}\,dx,\  v=-\frac{(1-x)^{n+1}}{n+1}$$

\begin{align*}
I_n
&=\left[-\frac{x^n(1-x)^{n+1}}{n+1}\right]_0^1+\frac{n}{n+1}\int_0^1 x^{n-1}(1-x)^{n+1}\,dx\\
&=\frac{n}{n+1}\int_0^1 x^{n-1}(1-x)^{n+1}\,dx
\end{align*}

Let $\ds A=\int_0^1 x^{n-1}(1-x)^n\,dx=\frac12 I_{n-1}$

$$\int_0^1 x^{n-1}(1-x)^{n+1}\,dx
=\int_0^1 x^{n-1}(1-x)^n(1-x)\,dx
=A-I_n$$

$$I_n=\frac{n}{n+1}(A-I_n)$$

$$I_n=\frac{n}{n+1}\left(\frac12 I_{n-1}-I_n\right)$$

$$\left(1+\frac{n}{n+1}\right)I_n=\frac{n}{2(n+1)}I_{n-1}$$

$$\therefore I_n=\frac{n}{2(2n+1)}I_{n-1}$$

\bigskip\hrule\bigskip

\textbf{(ii)}

$$I_0=\int_0^1 1\,dx=1$$

$$I_n=\frac{n}{2(2n+1)}I_{n-1}$$

\begin{align*}
I_n
&=\frac{n}{2(2n+1)}\cdot\frac{n-1}{2(2n-1)}\cdots\frac{1}{2\cdot3}\cdot I_0\\
&=\frac{n!}{2^n\cdot3\cdot5\cdots(2n+1)}\\
&=\frac{n!}{2^n\cdot\frac{(2n+1)!}{2^n n!}}=\frac{(n!)^2}{(2n+1)!}
\end{align*}


\bigskip\hrule\bigskip

\textbf{(iii)} When $\ds x=\sin^2\theta, \ dx=2\sin\theta\cos\theta\,d\theta,\  x=0\Rightarrow \theta=0,\  x=1\Rightarrow \theta=\frac{\pi}{2}$

\begin{align*}
I_{\frac12}
&=\int_0^1 x^{1/2}(1-x)^{1/2}\,dx\\
&=\int_0^{\pi/2}\sin\theta\cos\theta\cdot2\sin\theta\cos\theta\,d\theta\\
&=2\int_0^{\pi/2}\sin^2\theta\cos^2\theta\,d\theta\\
&=\frac12\int_0^{\pi/2}\sin^2 2\theta\,d\theta\\
&=\frac14\int_0^{\pi/2}(1-\cos4\theta)\,d\theta\\
&=\frac14\left[\theta-\frac14\sin4\theta\right]_0^{\pi/2}\\
&=\frac{\pi}{8}
\end{align*}

$$I_{\frac32}=\frac{\frac32}{2(2\cdot\frac32+1)}I_{\frac12}=\frac{3}{16}\cdot\frac{\pi}{8}=\frac{3\pi}{128}$$
