\[ \begin{aligned} \mathrm{Corr}(Z_1,Z_2) &=\frac{\mathrm{Cov}(Z_1,Z_2)}{\sqrt{\mathrm{Var}(Z_1)\mathrm{Var}(Z_2)}}\\ &=\mathrm{Cov}(Z_1,Z_2)\\ &=\mathrm{E}\big((Z_1-\mu_{Z_1})(Z_2-\mu_{Z_2})\big)\\ &=E(Z_1Z_2)\\ &=0 \end{aligned} \]
\[\mathrm{E}(Y_2)=\rho_{12}\mathrm{E}(Z_1)+\sqrt{1-\rho_{12}^2}\,\mathrm{E}(Z_2)=0\]
\[\mathrm{Var}(Y_2)=\rho_{12}^2\mathrm{Var}(Z_1)+(1-\rho_{12}^2)\mathrm{Var}(Z_2)=\rho_{12}^2+1-\rho_{12}^2=1\]
\[ \begin{aligned} \mathrm{Corr}(Y_1,Y_2) &=\frac{\mathrm{Cov}(Y_1,Y_2)}{\sqrt{\mathrm{Var}(Y_1)\mathrm{Var}(Y_2)}}\\ &=\mathrm{Cov}(Y_1,Y_2)\\ &=\mathrm{Cov}\left( Z_1,\rho_{12}Z_1+\sqrt{1-\rho_{12}^2}\,Z_2 \right)\\ &=\rho_{12}\mathrm{Var}(Z_1) +\sqrt{1-\rho_{12}^2}\,\mathrm{Cov}(Z_1,Z_2)\\ &=\rho_{12} \end{aligned} \]
\[ \begin{aligned} \rho_{13} &=\mathrm{Corr}(Y_1,Y_3)\\ &=\frac{\mathrm{Cov}(Y_1,Y_3)}{\sqrt{\mathrm{Var}(Y_1)\mathrm{Var}(Y_3)}}\\ &=\mathrm{Cov}(Y_1,Y_3)\\ &=E(Y_1Y_3)\\ &=E\big(Z_1(aZ_1+bZ_2+cZ_3)\big)\\ &=aE(Z_1^2)+bE(Z_1Z_2)+cE(Z_1Z_3)\\ &=a \end{aligned} \]
\[ \begin{aligned} \rho_{23} &=\mathrm{Corr}(Y_2,Y_3)\\ &=\mathrm{Cov}\left( \rho_{12}Z_1+\sqrt{1-\rho_{12}^2}\,Z_2, aZ_1+bZ_2+cZ_3 \right)\\ &=\rho_{12}a+\sqrt{1-\rho_{12}^2}\,b \end{aligned} \]
\[b=\frac{\rho_{23}-\rho_{12}a} {\sqrt{1-\rho_{12}^2}}\]
\[ \begin{aligned} \mathrm{Var}(Y_3) &=\mathrm{Var}(aZ_1+bZ_2+cZ_3)\\ &=a^2\mathrm{Var}(Z_1)+b^2\mathrm{Var}(Z_2)+c^2\mathrm{Var}(Z_3)\\ &=a^2+b^2+c^2=1 \end{aligned} \]
\[ c=\sqrt{1-a^2-b^2}=\sqrt{ 1-\rho_{13}^2 -\frac{(\rho_{23}-\rho_{12}\rho_{13})^2} {1-\rho_{12}^2} }\]

For \(i=1,2,3\)

\[ \boxed{X_i=\mu_i+\sigma_iY_i} \]
\[\mathrm{E}(X_i)=\mu_i+\sigma_i\mathrm{E}(Y_i)=\mu_i\]
\[\mathrm{Var}(X_i)=\sigma_i^2\mathrm{Var}(Y_i)=\sigma_i^2\]

For \(i\neq j\)

\[ \begin{aligned} \mathrm{Corr}(X_i,X_j) &=\mathrm{Corr}(\mu_i+\sigma_iY_i,\mu_j+\sigma_jY_j)\\ &=\mathrm{Corr}(Y_i,Y_j)\\ &=\rho_{ij} \end{aligned} \]
