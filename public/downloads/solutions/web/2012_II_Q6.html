\[ \begin{aligned} BD^2&=a^2+d^2-2ad\cos\theta\\ BD^2&=b^2+c^2-2bc\cos(180^\circ-\theta) \end{aligned} \]
\[ a^2+d^2-2ad\cos\theta=b^2+c^2+2bc\cos\theta \]
\[ \boxed{\cos\theta=\frac{a^2+d^2-b^2-c^2}{2(ad+bc)}} \]
\[ \begin{aligned} Q_1&=\frac12ad\sin\theta\\ Q_2&=\frac12bc\sin(180^\circ-\theta) \end{aligned} \]
\[ Q=Q_1+Q_2=\frac12(ad+bc)\sin\theta \]
\[ \boxed{\sin\theta=\frac{2Q}{ad+bc}} \]
\[ \because\sin^2\theta+\cos^2\theta=1 \]
\[ \therefore \left(\frac{2Q}{ad+bc}\right)^2+\left(\frac{a^2+d^2-b^2-c^2}{2(ad+bc)}\right)^2=1 \]
\[ \implies \boxed{16Q^2=4(ad+bc)^2-(a^2+d^2-b^2-c^2)^2} \]
\[ \begin{aligned} 16Q^2 &=\left(2(ad+bc)+a^2+d^2-b^2-c^2\right)\left(2(ad+bc)-a^2-d^2+b^2+c^2\right)\\ &=\left((a+d)^2-(b-c)^2\right)\left((b+c)^2-(a-d)^2\right)\\ &=(a+d-b+c)(a+d+b-c)(b+c-a+d)(b+c+a-d)\\ &=16(s-a)(s-b)(s-c)(s-d) \end{aligned} \]
\[ \boxed{Q^2=(s-a)(s-b)(s-c)(s-d)} \]

For a triangle with sides \(\displaystyle a,b,c\), let \(\displaystyle d=0\)

\[ Q^2=s(s-a)(s-b)(s-c) \]
\[ Q=\sqrt{s(s-a)(s-b)(s-c)} \]
