(i)

Let \(u=1-x, \quad dx=-du, \qquad \begin{array}{c|cc} x & 0 & 1\\ \hline u & 1 & 0 \end{array} \)

\[ \begin{aligned} \mathrm{b}(p,q) &=\int_0^1x^{p-1}(1-x)^{q-1}\,dx\\ &=\int_0^1u^{q-1}(1-u)^{p-1}\,du\\ &=\mathrm{b}(q,p) \end{aligned} \]

(ii)

\[ \begin{aligned} \mathrm{b}(p,q)-\mathrm{b}(p,q+1)&=\int_0^1x^{p-1}(1-x)^{q-1}\left[1-(1-x)\right]\,dx\\ &=\int_0^1x^p(1-x)^{q-1}\,dx\\ &=\mathrm{b}(p+1,q) \end{aligned} \]
\[ \begin{aligned} \mathrm{b}(p+1,p)&=\mathrm{b}(p,p)-\mathrm{b}(p,p+1)\\ &=\mathrm{b}(p,p)-\mathrm{b}(p+1,p)\\ &=\frac{1}{2}\mathrm{b}(p,p) \end{aligned} \]

(iii) Let \(x=\sin^2\theta, \qquad 1-x=\cos^2\theta,\qquad dx=2\sin\theta\cos\theta\,d\theta,\qquad \begin{array}{c|cc} x & 0 & 1\\ \hline \theta & 0 & \frac{\pi}{2} \end{array} \)

\[ \begin{aligned} \mathrm{b}(p,q) &=\int_0^1x^{p-1}(1-x)^{q-1}\,dx\\ &=\int_0^{\frac{\pi}{2}} (\sin^2\theta)^{p-1}(\cos^2\theta)^{q-1} (2\sin\theta\cos\theta)\,d\theta\\ &=2\int_0^{\frac{\pi}{2}} (\sin\theta)^{2p-1}(\cos\theta)^{2q-1}\,d\theta. \end{aligned} \]
\[ \begin{aligned} \mathrm{b}(p,p) &=2\int_0^{\frac{\pi}{2}} (\sin\theta\cos\theta)^{2p-1}\,d\theta\\ &=2\int_0^{\frac{\pi}{2}} \left(\frac12\sin2\theta\right)^{2p-1}\,d\theta\\ &=\frac{2}{2^{2p-1}} \int_0^{\frac{\pi}{2}}(\sin2\theta)^{2p-1}\,d\theta\\ &=\frac{1}{2^{2p-1}}\int_0^\pi(\sin u)^{2p-1}\,du\\ &=\frac{2}{2^{2p-1}} \int_0^{\frac{\pi}{2}}(\sin u)^{2p-1}\,du \end{aligned} \]

When \(q=\frac{1}{2}\)

\[\mathrm{b}\left(p,\frac12\right)=2\int_0^{\frac{\pi}{2}}(\sin u)^{2p-1}\,du\]

Hence,

\[ \boxed{ \mathrm{b}(p,p) =\frac{1}{2^{2p-1}}\mathrm{b}\left(p,\frac12\right) } \]

(iv)

Let \(t=\dfrac{x}{1-x},\qquad \displaystyle dx=\frac{1}{(1+t)^2}\,dt,\qquad \begin{array}{c|cc} x & 0 & 1\\ \hline t & 0 & \infty \end{array} \)

\[ \begin{aligned} \mathrm{b}(p,q) &=\int_0^1x^{p-1}(1-x)^{q-1}\,dx\\ &=\int_0^\infty \left(\frac{t}{1+t}\right)^{p-1} \left(\frac{1}{1+t}\right)^{q-1} \frac{1}{(1+t)^2}\,dt\\ &=\int_0^\infty \frac{t^{p-1}}{(1+t)^{p+q}}\,dt \end{aligned} \]

Hence,

\[ \boxed{ \mathrm{b}(p,q)=\int_0^\infty \frac{t^{p-1}}{(1+t)^{p+q}}\,dt } \]

(v)

\[ \begin{aligned} \mathrm{b}\left(\frac52,\frac72\right)&=\int_0^\infty\frac{t^{\frac32}}{(1+t)^6}\,dt &&\text{from part (iv)} \end{aligned} \]
\[\mathrm{b}\left(\frac52,\frac52\right) -\mathrm{b}\left(\frac52,\frac72\right)=\mathrm{b}\left(\frac72,\frac52\right)=\mathrm{b}\left(\frac52,\frac72\right)\]
\[ \begin{aligned} \mathrm{b}\left(\frac52,\frac72\right) &=\frac12\mathrm{b}\left(\frac52,\frac52\right)\\ &=\frac12\cdot\frac{1}{2^4} \mathrm{b}\left(\frac52,\frac12\right) &&\text{from part (iii)}\\ &=\frac{1}{2^5}\mathrm{b}\left(\frac52,\frac12\right) \end{aligned} \]
\[\mathrm{b}\left(\frac32,\frac12\right)=2\int_0^{\frac{\pi}{2}}\sin^2\theta\,d\theta=\frac{\pi}{2}\qquad \text{from part (iii)}\]
\[\mathrm{b}\left(\frac32,\frac32\right)=\frac{1}{2^2}\mathrm{b}\left(\frac32,\frac12\right)=\frac{\pi}{8}\qquad \text{from part (iii)}\]
\[\mathrm{b}\left(\frac52,\frac12\right) =\mathrm{b}\left(\frac32,\frac12\right) -\mathrm{b}\left(\frac32,\frac32\right) =\frac{\pi}{2}-\frac{\pi}{8} =\frac{3\pi}{8} \qquad \text{from part (ii)}\]
\[\int_0^\infty\frac{t^{\frac32}}{(1+t)^6}\,dt=\frac{1}{2^5}\cdot\frac{3\pi}{8}=\boxed{\frac{3\pi}{256}}\]
