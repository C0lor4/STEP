\[ \begin{aligned} \frac{E_A}{K} &=1\cdot\binom41pq^3 +4\left(1-q^4-\binom41pq^3\right)\\ &=4pq^3+4(1-q^4-4pq^3)\\ &=4p(1+q+q^2-2q^3) \end{aligned} \]

Basingstoke

\[ \begin{aligned} \frac{E_B}{K} &=1\cdot\binom61pq^5 +2\cdot\binom62p^2q^4 +6\left(1-q^6-\binom61pq^5-\binom62p^2q^4\right)\\ &=6pq^5+30p^2q^4+6(1-q^6-6pq^5-15p^2q^4)\\ &=6-60q^4+90q^5-36q^6\\ &=6(1-q)(1+q+q^2+q^3-9q^4+6q^5) \end{aligned} \]
\[ E_B=6Kp(1+q+q^2+q^3-9q^4+6q^5) \]
\[ E_A=\frac23E_B \]
\[ 4Kp(1+q+q^2-2q^3) =\frac23\cdot6Kp(1+q+q^2+q^3-9q^4+6q^5) \]

\[ E_A=E_B=0 \]
