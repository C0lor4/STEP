\(n\) 3 4 5 6 7 8 9 10
\(F_n\) 2 3 5 8 13 21 34 55

(i)

\[ F_i=F_{i-1}+F_{i-2}\lt 2F_{i-1} \]
\[ \therefore \frac1{F_i}>\frac1{2F_{i-1}} \]

For \(\displaystyle i\geq 4\),

\[ \frac1{F_i}>\frac1{2F_{i-1}} \]
\[ \frac1{F_4}>\frac1{2F_3},\quad\frac1{F_5}>\frac1{2^2F_3},\quad\frac1{F_6}>\frac1{2^3F_3},\ \ldots \]
\[ S>1+1+\frac12+\frac1{2F_3}\left(1+\frac12+\frac14+\cdots\right)=3 \]

For \(\displaystyle i\geq 3\),

\[ F_{i+2}=F_{i+1}+F_i>2F_i \]
\[ \frac1{F_{i+2}}\lt\frac1{2F_i} \]
\[ \begin{aligned} S &=1+1+\left(\frac1{F_3}+\frac1{F_5}+\cdots\right) +\left(\frac1{F_4}+\frac1{F_6}+\cdots\right)\\ S&\lt 1+1+\frac1{F_3}\left(1+\frac12+\frac14+\cdots\right) +\frac1{F_4}\left(1+\frac12+\frac14+\cdots\right)\\ S&\lt 3\frac23 \end{aligned} \]

(ii)

\[ \begin{aligned} S&>1+1+\frac12+\frac13+\frac15+\frac18+\frac1{13}\\ &=\frac{5047}{1560}>3.2 \end{aligned} \]
\[ \begin{aligned} S &=1+1+\frac12+\frac13+\frac15 +\left(\frac1{F_6}+\frac1{F_8}+\cdots\right) +\left(\frac1{F_7}+\frac1{F_9}+\cdots\right)\\ &\lt\frac{91}{30}+\frac2{F_6}+\frac2{F_7}\approx3.03+0.24+0.15=3.42 \end{aligned} \]
\[ \because 3.42\lt 3.5 \]
\[ \therefore S\lt 3.5 \]
