\[ \begin{aligned} 5x^2+2y^2-6xy+4x-4y&=a(x-y+2)^2+b(cx+y)^2+d\\ &=(a+bc^2)x^2+(a+b)y^2+(-2a+2bc)xy +4ax-4ay+4a+d \end{aligned} \]
\[ \begin{cases} a+bc^2=5\\ a+b=2\\ -2a+2bc=-6\\ 4a=4\\ 4a+d=0 \end{cases} \implies \begin{cases} a=1\\ b=1\\ c=-2\\ d=-4 \end{cases} \]
\[5x^2+2y^2-6xy+4x-4y=(x-y+2)^2+(y-2x)^2-4\]
\[ \begin{cases} (x-y+2)^2+(y-2x)^2=13\\ (6x^2+3y^2-8xy+8x-8y)-(5x^2+2y^2-6xy+4x-4y)=5 \end{cases} \]
\[ \implies \begin{cases} (x-y+2)^2+(y-2x)^2=13\\ (x-y+2)^2=9 \end{cases} \implies \begin{cases} x-y+2=\pm3\\ y-2x=\pm2 \end{cases} \]

Let \(\displaystyle u=x-y+2,\ v=y-2x\).

\[ \begin{cases} u=\pm3\\ v=\pm2 \end{cases} \implies \begin{cases} x=2-u-v\\ y=4-2u-v \end{cases} \]
\[ \begin{array}{c|c|c} u&v&(x,y)\\ \hline 3&2&(-3,-4)\\ 3&-2&(1,0)\\ -3&2&(3,8)\\ -3&-2&(7,12) \end{array} \]
