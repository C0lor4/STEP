(i)

\[ \because n>1,\ k\geq 1 \]
\[ \therefore(k+1)n^{k+1}>kn^k \]
\[ \frac1{(k+1)n^{k+1}}<\frac1{kn^k} \]

When \(\displaystyle x=\frac1n\),

\[ \ln\left(1+\frac1n\right)=\frac1n-\frac1{2n^2}+\frac1{3n^3}-\frac1{4n^4}+\cdots \]
\[ \begin{aligned} \ln\left(1+\frac1n\right) &=\frac1n-\left(\frac1{2n^2}-\frac1{3n^3}\right)-\left(\frac1{4n^4}-\frac1{5n^5}\right)-\cdots\\ <\frac1n \end{aligned} \]
\[ \implies \ln\left(1+\frac1n\right)^n<1 \]
\[ \left(1+\frac1n\right)^n<e \]

(ii) Let \(\displaystyle u=\frac1{2y},\ y>\frac12\implies 0<u<1\)

\[ \begin{aligned} \ln\left(\frac{2y+1}{2y-1}\right) &=\ln\left(\frac{1+u}{1-u}\right)\\ &=\ln(1+u)-\ln(1-u)\\ &=\left(u-\frac{u^2}{2}+\frac{u^3}{3}-\cdots\right) -\left(-u-\frac{u^2}{2}-\frac{u^3}{3}-\cdots\right)\\ &=2\left(u+\frac{u^3}{3}+\frac{u^5}{5}+\cdots\right)>\frac1y \end{aligned} \]

When \(\displaystyle y=n+\frac12\)

\[ \frac{2y+1}{2y-1}=\frac{2n+2}{2n}=1+\frac1n \]
\[ \ln\left(1+\frac1n\right)>\frac1{n+\frac12} \]
\[ \implies e<\left(1+\frac1n\right)^{n+\frac12} \]

(iii)

\[ e<\left(1+\frac1n\right)^{n+\frac12} =\left(1+\frac1n\right)^n\sqrt{1+\frac1n} \]
\[ \frac{e}{\sqrt{1+\frac1n}}<\left(1+\frac1n\right)^n<e\qquad \text{Parts (i)(ii)} \]
\[ \lim_{n\to\infty}\frac{e}{\sqrt{1+\frac1n}}=e<\lim_{n\to\infty}\left(1+\frac1n\right)^n<e \]
\[ \therefore \left(1+\frac1n\right)^n\to e \]
