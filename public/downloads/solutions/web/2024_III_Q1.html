(i)

\[ \begin{aligned} \frac1{r+1}-\frac1r+\frac1{r^2} &=\frac{r^2-r(r+1)+(r+1)}{r^2(r+1)}\\ &=\frac{r^2-r^2-r+r+1}{r^2(r+1)}\\ &=\frac1{r^2(r+1)} \end{aligned} \]
\[ \begin{aligned} \sum_{r=1}^N\frac1{r^2(r+1)} &=\sum_{r=1}^N\left(\frac1{r+1}-\frac1r\right)+\sum_{r=1}^N\frac1{r^2}\\ &=\left(\frac1{N+1}-1\right)+S_N\\ &=\sum_{r=1}^N\frac1{r^2}-1+\frac1{N+1} \end{aligned} \]

As \(\displaystyle N\to\infty\), \(\displaystyle \frac1{N+1}\to0\) and \(\displaystyle S_N\to\frac16\pi^2\)

\[ \implies \boxed{\sum_{r=1}^\infty\frac1{r^2(r+1)}=\frac16\pi^2-1} \]

(ii)

\[ \frac1{r^2(r+1)}-\frac1{r(r+1)(r+2)} =\frac{(r+2)-r}{r^2(r+1)(r+2)} =\frac2{r^2(r+1)(r+2)} \]
\[ \frac1{r(r+1)(r+2)}=\frac12\left(\frac1{r(r+1)}-\frac1{(r+1)(r+2)}\right) \]
\[ \implies\sum_{r=1}^N\frac1{r(r+1)(r+2)} =\frac12\left(\frac12-\frac1{(N+1)(N+2)}\right) \]
\[ \begin{aligned} \sum_{r=1}^N\frac1{r^2(r+1)(r+2)} &=\frac12\sum_{r=1}^N\frac1{r^2(r+1)}-\frac12\sum_{r=1}^N\frac1{r(r+1)(r+2)}\\ &=\frac12\left(S_N-1+\frac1{N+1}\right) -\frac14\left(\frac12-\frac1{(N+1)(N+2)}\right)\\ &=\frac12S_N-\frac58+\frac1{2(N+1)}+\frac1{4(N+1)(N+2)}\\ &=\frac12S_N-\frac58+\frac{2N+5}{4(N+1)(N+2)} \end{aligned} \]

As \(\displaystyle N\to\infty\), \(\displaystyle \frac{2N+5}{4(N+1)(N+2)}\to0\)

\[ \implies\boxed{\sum_{r=1}^\infty\frac1{r^2(r+1)(r+2)}=\frac1{12}\pi^2-\frac58} \]

(iii)

\[ \frac1{r^2(r+1)^2}=\left(\frac1r-\frac1{r+1}\right)^2 =\frac1{r^2}+\frac1{(r+1)^2}-\frac2{r(r+1)} \]
\[ \because \frac2{r^2(r+1)}=\frac2{r^2}-\frac2{r(r+1)} \]
\[ \therefore \frac2{r^2(r+1)}-\frac1{r^2(r+1)^2}=\frac1{r^2}-\frac1{(r+1)^2} \]
\[ \sum_{r=1}^N\left(\frac1{r^2}-\frac1{(r+1)^2}\right)=1-\frac1{(N+1)^2}\ \implies\ 1 \]

Both \(\displaystyle \sum\frac2{r^2(r+1)}\) and \(\displaystyle \sum\frac1{r^2(r+1)^2}\) converge, thus

\[ \boxed{\sum_{r=1}^\infty\frac1{r^2(r+1)^2} =\sum_{r=1}^\infty\frac2{r^2(r+1)}-1} \]
