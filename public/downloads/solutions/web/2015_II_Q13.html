(i)

\[ C= \begin{cases} ky, & X\leq y\\ ky+a(X-y), & X>y \end{cases} \]
\[ \begin{aligned} \mathrm{E}(C) &=\int_0^y ky\lambda e^{-\lambda x}\,dx +\int_y^\infty\left[ky+a(x-y)\right]\lambda e^{-\lambda x}\,dx\\ &=ky\int_0^\infty\lambda e^{-\lambda x}\,dx +a\int_y^\infty(x-y)\lambda e^{-\lambda x}\,dx\\ &=ky+a\left[-(x-y)e^{-\lambda x}-\frac1\lambda e^{-\lambda x}\right]_y^\infty\\ &=ky+\frac{a}{\lambda}e^{-\lambda y} \end{aligned} \]
\[ \frac{d}{dy}\mathrm{E}(C)=k-ae^{-\lambda y} \quad \frac{d^2}{dy^2}\mathrm{E}(C)=a\lambda e^{-\lambda y}>0 \]
\[ k-ae^{-\lambda y}=0 \quad\Longrightarrow\quad y=\frac{1}{\lambda}\ln\left(\frac{a}{k}\right) \]
\[ \boxed{ \therefore y= \begin{cases} 0, & \quad\dfrac{a}{k}\leq1\\[6pt] \dfrac{1}{\lambda}\ln\left(\dfrac{a}{k}\right), & \quad \dfrac{a}{k}>1 \end{cases} } \]

(ii) Let \(u=x-y\)

\[ \begin{aligned} \mathrm{E}(C^2) &=k^2y^2+ \int_y^\infty \left[2aky(x-y)+a^2(x-y)^2\right] \lambda e^{-\lambda x}\,dx\\ &=k^2y^2+e^{-\lambda y} \int_0^\infty \left(2akyu+a^2u^2\right) \lambda e^{-\lambda u}\,du\\ &=k^2y^2+e^{-\lambda y} \left( 2aky\int_0^\infty u\lambda e^{-\lambda u}\,du +a^2\int_0^\infty u^2\lambda e^{-\lambda u}\,du \right)\\ &=k^2y^2+e^{-\lambda y} \left( \frac{2aky}{\lambda} +a^2\left[-u^2e^{-\lambda u}\right]_0^\infty +2a^2\int_0^\infty ue^{-\lambda u}\,du \right)\\ &=k^2y^2+\left(\frac{2aky}{\lambda}+\frac{2a^2}{\lambda^2}\right)e^{-\lambda y} \end{aligned} \]
\[ \begin{aligned} \mathrm{Var}(C) &=\mathrm{E}(C^2)-\left(\mathrm{E}(C)\right)^2\\ &=k^2y^2+\left(\frac{2aky}{\lambda}+\frac{2a^2}{\lambda^2}\right)e^{-\lambda y} -\left(ky+\frac{a}{\lambda}e^{-\lambda y}\right)^2\\ &=\boxed{\frac{a^2}{\lambda^2} \left(2e^{-\lambda y}-e^{-2\lambda y}\right)} \end{aligned} \]
\[ \begin{aligned} \frac{d}{dy}\mathrm{Var}(C) &=\frac{2a^2}{\lambda} e^{-\lambda y}\left(e^{-\lambda y}-1\right) \end{aligned} \]

For \(y>0, \qquad e^{-\lambda y}<1 \quad\Longrightarrow\quad \frac{d}{dy}\mathrm{Var}(C)<0\)

\[ \boxed{\text{As y increase, Var(y) decreases}} \]
