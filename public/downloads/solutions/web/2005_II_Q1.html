(i)

\[ \begin{aligned} f(x)&=x^2e^{-x^2}\\ f'(x)&=2xe^{-x^2}-2x^3e^{-x^2}\\ &=2xe^{-x^2}(1-x^2) \end{aligned} \]

Since \(e^{-x^2}>0\)

\[ f'(x)=0\iff x(1-x^2)=0 \]
\[ \boxed{x=0,\ \pm1} \]

(ii)

\[ \begin{aligned} \frac{d}{dx}\left(P(x)e^{-x^2}\right) &=P'(x)e^{-x^2}-2xP(x)e^{-x^2}\\ &=e^{-x^2}\left(P'(x)-2xP(x)\right) \end{aligned} \]
\[ \begin{aligned} P'(x)-2xP(x)&=x(x^2-a^2)(x^2-b^2)\\ &=x^5-(a^2+b^2)x^3+a^2b^2x \end{aligned} \]

Assume \(\displaystyle \quad P(x)=-\frac12x^4+px^2+q\)

\[ \begin{aligned} P'(x)-2xP(x)&=-2x^3+2px-2x\left(-\frac12x^4+px^2+q\right)\\ &=x^5-(2p+2)x^3+(2p-2q)x \end{aligned} \]
\[ \left\{ \begin{aligned} 2p+2&=a^2+b^2\\ 2p-2q&=a^2b^2 \end{aligned} \right. \quad\Longrightarrow\quad \left\{ \begin{aligned} p&=\frac{a^2+b^2}{2}-1\\ q&=-1+\frac{a^2+b^2}{2}-\frac{a^2b^2}{2} \end{aligned} \right. \]
\[ \boxed{P(x)=-\frac12x^4+\left(\frac{a^2+b^2}{2}-1\right)x^2-1+\frac{a^2+b^2}{2}-\frac{a^2b^2}{2}} \]
