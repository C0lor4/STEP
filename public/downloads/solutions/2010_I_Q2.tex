$$y=\frac{x-a}{x-b}e^x$$

\begin{align*}
\frac{dy}{dx}
&=e^x\left(\frac{x-a}{x-b}+\frac{a-b}{(x-b)^2}\right)\\
&=\frac{e^x}{(x-b)^2}\bigl((x-a)(x-b)+a-b\bigr)\\
&=\frac{e^x}{(x-b)^2}\bigl(x^2-(a+b)x+ab+a-b\bigr)
\end{align*}

\begin{align*}
\Delta >0\implies \bigl(a+b\bigr)^2-4(ab+a-b)&>0\\
(a-b)(a-b-4)&>0
\end{align*}

$$a-b<0\quad\text{or}\quad a-b>4$$
\bigskip\hrule\bigskip

\textbf{(i)} 

\begin{align*}
x^2-\frac12x-\frac12&=0\\
2x^2-x-1&=0\\
(2x+1)(x-1)&=0
\end{align*}
$$x=-\frac12,\quad x=1$$

Vertical asymptote: $\ds x=\frac12$
$$\therefore -\frac12<\frac12<1$$

$$
\renewcommand{\arraystretch}{1.4}
\begin{array}{c!{\vrule width 1.2pt}c|c|c|c|c|c|c}
x&-\infty&-\frac12&0&\frac12^-&\frac12^+&1&\infty\\[4pt]
\hline
\rule[5pt]{0pt}{14pt}y&0&\frac{e^{-1/2}}{2}&0&-\infty&\infty&2e&\infty
\end{array}
$$
